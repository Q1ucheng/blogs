    STM32 HAL库 GPIO 输入输出控制¶

    主要体会了“面向接口编程”的思想。

    .h (设计接口) \(\rightarrow\) .c (实现细节) \(\rightarrow\) main.c (组装业务)

    第一步：编写头文件 (bsp_led.h / bsp_key.h)¶

    • 不用关心 LED 接在哪个引脚，我只关心我能对它做什么。
    • 动作：
      • 定义“契约”：比如我需要 LED_R_ON（红灯亮）、KEY_Scan（读按键）。
      • 这一步写完，我就确定了对外提供的“遥控器按钮”。

    第二步：编写源文件 (bsp_led.c / bsp_key.c)¶

    • 查原理图，把刚才定义的接口和具体的硬件连起来。
    • 动作：
      • 查阅原理图，确定 Pin（如 PB5）和电平逻辑（低电平亮）。
      • 配置 GPIO_InitTypeDef（Pin, Mode, Pull, Speed）。
      • 注意使能时钟 (__HAL_RCC_xxx_CLK_ENABLE)！

    /**
  * @file    bsp_led.h
  * @brief   [接口定义] LED操作宏与初始化声明
  */

#ifndef __BSP_LED_H
#define __BSP_LED_H

#include "stm32f1xx_hal.h"

/* 
 * 这里的宏定义对应原理图逻辑：
 * LED负极接GPIO -> 输出 RESET(0) 时与电源形成压差 -> 灯亮 
 * 使用 do-while(0) 是为了防止在 if 语句中宏展开出错
 */

/* 红色 LED 控制接口 */
#define LED_R_ON      do{ HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_RESET); }while(0)
#define LED_R_OFF     do{ HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_SET);   }while(0)
#define LED_R_TOGGLE  do{ HAL_GPIO_TogglePin(GPIOB, GPIO_PIN_5);                }while(0)

/* 绿色 LED 控制接口 */
#define LED_G_ON      do{ HAL_GPIO_WritePin(GPIOB, GPIO_PIN_0, GPIO_PIN_RESET); }while(0)
#define LED_G_OFF     do{ HAL_GPIO_WritePin(GPIOB, GPIO_PIN_0, GPIO_PIN_SET);   }while(0)
#define LED_G_TOGGLE  do{ HAL_GPIO_TogglePin(GPIOB, GPIO_PIN_0);                }while(0)

/* 蓝色 LED 控制接口 */
#define LED_B_ON      do{ HAL_GPIO_WritePin(GPIOB, GPIO_PIN_1, GPIO_PIN_RESET); }while(0)
#define LED_B_OFF     do{ HAL_GPIO_WritePin(GPIOB, GPIO_PIN_1, GPIO_PIN_SET);   }while(0)
#define LED_B_TOGGLE  do{ HAL_GPIO_TogglePin(GPIOB, GPIO_PIN_1);                }while(0)

/* 对外公开的初始化函数 */
void LED_GPIO_Init(void);

#endif /* __BSP_LED_H */

    /**
  * @file    bsp_led.c
  * @brief   [底层实现] LED GPIO的具体配置
  */

#include "bsp_led.h"

/**
  * @brief  初始化 LED 对应的 GPIOB 引脚
  */
void LED_GPIO_Init(void)
{
    /* 定义配置结构体，必须初始化为0以防垃圾值 */
    GPIO_InitTypeDef LED_GPIO_Init = {0};

    /* 【关键步骤】开启 GPIOB 时钟。不开时钟，后续操作全无效！ */
    __HAL_RCC_GPIOB_CLK_ENABLE();

    /* 配置参数 */
    /* Mode: 推挽输出 (Push-Pull)。这是驱动LED的标准模式，电流驱动能力强 */
    LED_GPIO_Init.Mode = GPIO_MODE_OUTPUT_PP; 

    /* Pin: 同时配置 PB0, PB1, PB5 */
    LED_GPIO_Init.Pin = GPIO_PIN_0 | GPIO_PIN_1 | GPIO_PIN_5;

    /* Pull: 无上下拉。输出模式下电平由程序控制，通常不需要内部电阻 */
    LED_GPIO_Init.Pull = GPIO_NOPULL;

    /* Speed: 低速。人眼看不出纳秒级变化，低速可降低干扰 */
    LED_GPIO_Init.Speed = GPIO_SPEED_FREQ_LOW;

    /* 将配置写入寄存器 */
    HAL_GPIO_Init(GPIOB, &LED_GPIO_Init);

    /* 初始化后默认把灯关掉 (置高电平) */
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_0 | GPIO_PIN_1 | GPIO_PIN_5, GPIO_PIN_SET);
}

    /**
  * @file    bsp_key.c
  * @brief   [底层实现] 按键 GPIO 配置与阻塞式扫描
  */

#include "bsp_key.h"

/**
  * @brief  初始化按键引脚 (PA0 和 PC13)
  */
void KEY_GPIO_Init(void)
{
    GPIO_InitTypeDef KEY_GPIO_Init = {0};

    /* 【关键步骤】同时开启 GPIOA 和 GPIOC 时钟 */
    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOC_CLK_ENABLE();

    /* 配置参数 */
    /* Mode: 输入模式。用于读取电压 */
    KEY_GPIO_Init.Mode = GPIO_MODE_INPUT;

    /* Pull: 无上下拉。
     * !注意!：因为原理图上已经画了外部电阻，所以这里用 NOPULL。
     * 如果没外部电阻，这里必须设为 GPIO_PULLUP 或 PULLDOWN 防止浮空。
     */
    KEY_GPIO_Init.Pull = GPIO_NOPULL;

    /* Speed: 输入模式下该参数无效，但保持默认即可 */
    KEY_GPIO_Init.Speed = GPIO_SPEED_FREQ_LOW;

    /* 初始化 PA0 */
    KEY_GPIO_Init.Pin = GPIO_PIN_0;
    HAL_GPIO_Init(GPIOA, &KEY_GPIO_Init);

    /* 初始化 PC13 (复用上面的配置) */
    KEY_GPIO_Init.Pin = GPIO_PIN_13;
    HAL_GPIO_Init(GPIOC, &KEY_GPIO_Init);
}

/**
  * @brief  读取按键状态
  * @param  GPIOx: 端口
  * @param  GPIO_Pin: 引脚
  * @return 1:按下, 0:未按下
  */
uint8_t KEY_Scan(GPIO_TypeDef* GPIOx, uint16_t GPIO_Pin)
{
    /* 读取引脚电平。假设按下是高电平(SET) */
    if(HAL_GPIO_ReadPin(GPIOx, GPIO_Pin) == GPIO_PIN_SET)
    {
        /* 消抖：如果按键一直按着，就死循环等待，防止一次按下被误判为多次 */
        /* !注意!：这是阻塞代码，实际项目中建议改用定时器中断或状态机 */
        while (HAL_GPIO_ReadPin(GPIOx, GPIO_Pin) == GPIO_PIN_SET);

        return KEY_ON;  // 确认按下
    }
    else
    {
        return KEY_OFF;
    }
}

    /**
  * @file    main.c
  * @brief   [业务层] 包含主循环
  */

#include "main.h"
#include "./led/bsp_led.h"
#include "./key/bsp_key.h"

/* 时钟配置函数声明 */
void SystemClock_Config(void);

int main(void)
{
    /* 1. HAL 库初始化 (复位外设, Flash接口, Systick等) */
    HAL_Init();

    /* 2. 配置系统时钟 (72MHz) */
    SystemClock_Config();

    /* 3. 硬件初始化 (调用我们写的BSP) */
    /* 这里不要再调用 MX_GPIO_Init 了，避免重复初始化 */
    LED_GPIO_Init();
    KEY_GPIO_Init();

    /* 4. 死循环 */
    while (1)
    {
        /* 业务逻辑：检测 PA0 控制 蓝灯 */
        if (KEY_Scan(GPIOA, GPIO_PIN_0) == KEY_ON)
        {
            LED_B_TOGGLE; // 直接调用接口，无需关心底层
        }

        /* 业务逻辑：检测 PC13 控制 绿灯 */
        if (KEY_Scan(GPIOC, GPIO_PIN_13) == KEY_ON)
        {
            LED_G_TOGGLE;
        }
    }
}
